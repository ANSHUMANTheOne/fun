function calculate(){ var n1 = document.getElementById("num1").value var n2 = document.getElementById("num2").value var sign = document.getElementById("mathSign").value var x = parseFloat(n1) var y = parseFloat(n2) var answer = 0 if(sign == "plus"){ answer = x + y } if(sign == "minus"){ answer = x - y } if(sign == "times"){ answer = x * y } if(sign == "divide"){ answer = x / y } document.getElementById("answerText").innerHTML = "result: " + answer }
